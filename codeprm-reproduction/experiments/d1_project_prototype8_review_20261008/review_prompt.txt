def range_alternate(p):
    groups = _range_groups(p["text"], p.get("separator", ","))
    expanded = [_range_expand(token, p.get("range_separator", "-")) for token in groups]
    return sorted(value for group in expanded for value in group)


def _range_groups(text, separator):
    return [token for token in text.strip().split(separator) if token]


def _range_expand(token, separator):
    bounds = [int(part) for part in token.split(separator)]
    if len(bounds) == 1:
        return bounds
    return list(range(min(bounds), max(bounds) + 1))

from collections import OrderedDict

def multimap_alternate(p):
    pairs, output = [], []
    for op in p["operations"]:
        name, key = op["op"], op.get("key")
        indexes = [i for i, pair in enumerate(pairs) if pair[0] == key]
        if name == "add":
            pairs.append([key, op["value"]]); result = None
        elif name == "set":
            pairs = [pair for pair in pairs if pair[0] != key]
            pairs.append([key, op["value"]]); result = None
        elif name == "get":
            result = pairs[indexes[-1]][1] if indexes else op.get("default")
        elif name == "getall":
            result = [pairs[i][1] for i in indexes]
        elif name == "poplast":
            result = pairs.pop(indexes[-1])[1] if indexes else {"error": "KeyError"}
        elif name == "delete":
            result = None if indexes else {"error": "KeyError"}
            pairs = [pair for pair in pairs if pair[0] != key]
        else:
            raise ValueError("unknown operation")
        output.append({"result": result, "items": [list(pair) for pair in pairs]})
    return output


def solve(payload):
    return multimap_alternate(payload)

def _cache_alternate(p, timed):
    capacity, state, expires, output = p["capacity"], OrderedDict(), {}, []
    if capacity < 0:
        raise ValueError("negative capacity")
    for op in p["operations"]:
        now = op["time"] if timed else 0
        for key in list(state):
            if timed and now >= expires[key]:
                del state[key]; del expires[key]
        key, name = op.get("key"), op["op"]
        if name == "put":
            if capacity == 0:
                result = {"error": "ValueError"}
            else:
                if key in state:
                    del state[key]
                elif len(state) >= capacity:
                    old, _ = state.popitem(last=False); expires.pop(old, None)
                state[key] = op["value"]
                if timed:
                    expires[key] = now + p["ttl"]
                result = None
        elif name == "get":
            if key not in state:
                result = {"error": "KeyError"}
            else:
                result = state[key]; state.move_to_end(key)
        elif name == "delete":
            if key not in state:
                result = {"error": "KeyError"}
            else:
                del state[key]; expires.pop(key, None); result = None
        elif name == "contains":
            result = key in state
        elif name == "clear":
            state.clear(); expires.clear(); result = None
        elif name == "snapshot":
            result = None
        else:
            raise ValueError("unknown operation")
        # A newly inserted entry with nonpositive TTL is already expired.
        for key in list(state):
            if timed and now >= expires[key]:
                del state[key]; del expires[key]
        output.append({"result": result, "items": [[k, state[k]] for k in sorted(state)], "size": len(state)})
    return output

def _cache_alternate(p, timed):
    capacity, state, expires, output = p["capacity"], OrderedDict(), {}, []
    if capacity < 0:
        raise ValueError("negative capacity")
    for op in p["operations"]:
        now = op["time"] if timed else 0
        for key in list(state):
            if timed and now >= expires[key]:
                del state[key]; del expires[key]
        key, name = op.get("key"), op["op"]
        if name == "put":
            if capacity == 0:
                result = {"error": "ValueError"}
            else:
                if key in state:
                    del state[key]
                elif len(state) >= capacity:
                    old, _ = state.popitem(last=False); expires.pop(old, None)
                state[key] = op["value"]
                if timed:
                    expires[key] = now + p["ttl"]
                result = None
        elif name == "get":
            if key not in state:
                result = {"error": "KeyError"}
            else:
                result = state[key]; state.move_to_end(key)
        elif name == "delete":
            if key not in state:
                result = {"error": "KeyError"}
            else:
                del state[key]; expires.pop(key, None); result = None
        elif name == "contains":
            result = key in state
        elif name == "clear":
            state.clear(); expires.clear(); result = None
        elif name == "snapshot":
            result = None
        else:
            raise ValueError("unknown operation")
        # A newly inserted entry with nonpositive TTL is already expired.
        for key in list(state):
            if timed and now >= expires[key]:
                del state[key]; del expires[key]
        output.append({"result": result, "items": [[k, state[k]] for k in sorted(state)], "size": len(state)})
    return output

            actual={'value':module.solve(payload)}

        passed=actual==case['expected'] and payload==before
        results.append({'case_id':case['case_id'],'pass':passed,'input_unchanged':payload==before,
                        'error':None if passed else {'expected':case['expected'],'actual':actual}})
    print(json.dumps({'cases':len(results),'passed':sum(row['pass'] for row in results),
                      'all_pass':all(row['pass'] for row in results),'results':results},ensure_ascii=False))

def alternate_source(function):
    collected = {}
    def collect(fn):
        if fn.__name__ in collected:
            return
        source = inspect.getsource(fn)
        collected[fn.__name__] = source
        used = {node.id for node in ast.walk(ast.parse(source)) if isinstance(node, ast.Name)}
        for name in used:
            dependency = getattr(task_specs, name, None)
            if inspect.isfunction(dependency) and dependency.__module__ == "task_specs":
                collect(dependency)
    collect(function)
    return ("from collections import OrderedDict\n\n" + "\n\n".join(collected.values())
            + f"\n\ndef solve(payload):\n    return {function.__name__}(payload)\n")

                result = cache[key]
            elif name == "delete":
                del cache[key]; result = None
            elif name == "contains":
                result = key in cache
            elif name == "clear":
                cache.clear(); result = None
            elif name == "snapshot":
                result = None
            else:
                raise ValueError("unknown operation")
        except (KeyError, ValueError) as exc:
            result = {"error": type(exc).__name__}
        size = cache.currsize
        items = [[k, values[k]] for k in sorted(values) if k in cache]
        output.append({"result": result, "items": items, "size": size})
    return output

def _cache_alternate(p, timed):
    capacity, state, expires, output = p["capacity"], OrderedDict(), {}, []
    if capacity < 0:
        raise ValueError("negative capacity")
    for op in p["operations"]:
        now = op["time"] if timed else 0
        for key in list(state):
            if timed and now >= expires[key]:
                del state[key]; del expires[key]
        key, name = op.get("key"), op["op"]
        if name == "put":
            if capacity == 0:
                result = {"error": "ValueError"}
            else:
                if key in state:
                    del state[key]
                elif len(state) >= capacity:
                    old, _ = state.popitem(last=False); expires.pop(old, None)
                state[key] = op["value"]
                if timed:
                    expires[key] = now + p["ttl"]
                result = None
        elif name == "get":
            if key not in state:
                result = {"error": "KeyError"}
            else:
                result = state[key]; state.move_to_end(key)
        elif name == "delete":
            if key not in state:
                result = {"error": "KeyError"}
            else:
                del state[key]; expires.pop(key, None); result = None
        elif name == "contains":
            result = key in state
        elif name == "clear":
            state.clear(); expires.clear(); result = None
        elif name == "snapshot":
            result = None
        else:
            raise ValueError("unknown operation")
        # A newly inserted entry with nonpositive TTL is already expired.
        for key in list(state):
            if timed and now >= expires[key]:
                del state[key]; del expires[key]
        output.append({"result": result, "items": [[k, state[k]] for k in sorted(state)], "size": len(state)})
    return output

def strings_reference(p):
    import confuse

    view = confuse.RootView([confuse.ConfigSource({"value": p["value"]})])
    try:

def strings_alternate(p):
    value = p["value"]
    if isinstance(value, str):
        return value.split() if p.get("split", True) else [value]
    if isinstance(value, list) and all(isinstance(v, str) for v in value):
        return value[:]
    raise ValueError("invalid string sequence")

def pairs_reference(p):
    import confuse
    view = confuse.RootView([confuse.ConfigSource({"value": p["value"]})])
    try:
        return [list(pair) for pair in view["value"].get(confuse.Pairs(default_value=p.get("default")))]
    except confuse.ConfigError as exc:
        raise ValueError("invalid pairs") from exc

def pairs_alternate(p):
    value = p["value"]
    output = []
    for item in value:
        if isinstance(item, str):
            output.append([item, p.get("default")])
        elif isinstance(item, dict) and len(item) == 1:
            key = next(iter(item)); output.append([key, item[key]])
        elif isinstance(item, list) and len(item) == 2:
            output.append(item[:])
        else:
            raise ValueError("invalid pairs")
    return output
