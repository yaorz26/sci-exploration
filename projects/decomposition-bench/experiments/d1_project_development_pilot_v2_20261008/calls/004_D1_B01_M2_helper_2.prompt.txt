import re

def _split_text(text, separator):
    # 使用正则表达式去除首尾空白并分割文本
    segments = re.split(r'\s*' + re.escape(separator) + r'\s*', text.strip())
    # 过滤掉空分段
    segments = [segment for segment in segments if segment]
    return segments
